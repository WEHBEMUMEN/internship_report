\section{Proper Orthogonal Decomposition (POD), Galerkin ROM, and ECSW Hyper-Reduction}
\label{sec:pod_rom}

This section details the reduced order modeling formulas, Singular Value Decomposition (SVD) basis truncation, Galerkin subspace projection, and Energy-Conserving Sampling and Weighting (ECSW) hyper-reduction used to achieve a 39x computational speedup.

\subsection{Snapshot Collection Matrix $\mathbf{S}$}

\subsubsection{Formula}
Full order model (FOM) displacement solutions are generated offline across parameter samples $\boldsymbol{\mu}_k \in \mathcal{D}$ and time instances $t_j \in [0, T]$, forming the snapshot matrix:
\begin{equation}
\mathbf{S} = \begin{bmatrix} \mathbf{u}(\boldsymbol{\mu}_1, t_1) & \mathbf{u}(\boldsymbol{\mu}_1, t_2) & \dots & \mathbf{u}(\boldsymbol{\mu}_{n_p}, t_{n_t}) \end{bmatrix} \in \mathbb{R}^{N \times n_s}
\end{equation}
where $N$ is the number of FOM degrees of freedom, and $n_s = n_p \times n_t$ is total snapshot count in reference space $\mathcal{V}_0(\hat{\Omega})$.

\subsection{Thin Singular Value Decomposition (SVD) \& Optimal Subspace}

\subsubsection{Formula}
The SVD of snapshot matrix $\mathbf{S}$ is:
\begin{equation}
\mathbf{S} = \mathbf{U} \boldsymbol{\Sigma} \mathbf{V}^T = \sum_{i=1}^{\min(N, n_s)} \sigma_i \mathbf{\phi}_i \mathbf{v}_i^T
\end{equation}
where:
\begin{itemize}
    \item $\mathbf{U} = [\mathbf{\phi}_1, \mathbf{\phi}_2, \dots, \mathbf{\phi}_N] \in \mathbb{R}^{N \times N}$ contains orthogonal POD basis modes.
    \item $\boldsymbol{\Sigma} = \text{diag}(\sigma_1, \sigma_2, \dots, \sigma_R) \in \mathbb{R}^{N \times n_s}$ contains singular values sorted in descending order ($\sigma_1 \ge \sigma_2 \ge \dots \ge \sigma_R \ge 0$).
    \item $\mathbf{V} \in \mathbb{R}^{n_s \times n_s}$ contains temporal modes.
\end{itemize}

\subsection{POD Basis Truncation \& Energy Ratio $\eta(r)$}

\subsubsection{Formula}
The reduced basis matrix $\boldsymbol{\Phi} \in \mathbb{R}^{N \times r}$ ($r \ll N$) is formed by selecting the first $r$ columns of $\mathbf{U}$:
\begin{equation}
\boldsymbol{\Phi} = \begin{bmatrix} \mathbf{\phi}_1 & \mathbf{\phi}_2 & \dots & \mathbf{\phi}_r \end{bmatrix}
\end{equation}
Optimal dimension $r$ is chosen based on cumulative energy ratio $\eta(r)$:
\begin{equation}
\eta(r) = \frac{\sum_{i=1}^r \sigma_i^2}{\sum_{j=1}^{n_s} \sigma_j^2} \ge 1 - \varepsilon_{\text{POD}}
\end{equation}
where typical truncation tolerance is $\varepsilon_{\text{POD}} = 10^{-4}$ ($99.99\%$ energy capture).

\subsection{Galerkin Subspace Projection (POD-ROM System)}

\subsubsection{Formula}
Subspace projection approximates displacement $\mathbf{u}(t) \in \mathbb{R}^N$ by reduced coordinates $\mathbf{q}(t) \in \mathbb{R}^r$:
\begin{equation}
\mathbf{u}(t) \approx \boldsymbol{\Phi} \mathbf{q}(t)
\end{equation}
Pre-multiplying the semi-discrete system by $\boldsymbol{\Phi}^T$ yields the Reduced Order Model (ROM):
\begin{equation}
\mathbf{M}_r \ddot{\mathbf{q}}(t) + \mathbf{C}_r \dot{\mathbf{q}}(t) + \mathbf{K}_r(\boldsymbol{\mu}) \mathbf{q}(t) = \mathbf{F}_r(t)
\end{equation}
where reduced operators are:
\begin{equation}
\mathbf{M}_r = \boldsymbol{\Phi}^T \mathbf{M} \boldsymbol{\Phi} \in \mathbb{R}^{r \times r}, \quad 
\mathbf{C}_r = \boldsymbol{\Phi}^T \mathbf{C} \boldsymbol{\Phi} \in \mathbb{R}^{r \times r}
\end{equation}
\begin{equation}
\mathbf{K}_r(\boldsymbol{\mu}) = \boldsymbol{\Phi}^T \mathbf{K}(\boldsymbol{\mu}) \boldsymbol{\Phi} \in \mathbb{R}^{r \times r}, \quad 
\mathbf{F}_r(t) = \boldsymbol{\Phi}^T \mathbf{F}(t) \in \mathbb{R}^r
\end{equation}

\subsection{Energy-Conserving Sampling and Weighting (ECSW) Hyper-Reduction}

\subsubsection{Formula}
ECSW approximates reduced stiffness matrix $\mathbf{K}_r(\boldsymbol{\mu})$ by a weighted sum over a small active element sub-mesh $\Omega_{\text{ECSW}} \subset \hat{\Omega}$:
\begin{equation}
\mathbf{K}_{r,\text{ECSW}}(\boldsymbol{\mu}) = \sum_{e \in \Omega_{\text{ECSW}}} w_e \, \boldsymbol{\Phi}_e^T \mathbf{K}_e(\boldsymbol{\mu}) \boldsymbol{\Phi}_e
\end{equation}
where $\boldsymbol{\Phi}_e \in \mathbb{R}^{2N_e \times r}$ are element POD modes, $w_e > 0$ are non-negative element weights, and $E_{\text{ECSW}} = |\Omega_{\text{ECSW}}| \ll E$.

\subsubsection{Sparse Weights Optimization via Non-Negative Least Squares (NNLS)}
Optimal weights $\mathbf{w} = [w_1, w_2, \dots, w_E]^T$ minimize internal virtual work force residuals:
\begin{equation}
\min_{\mathbf{w} \ge 0} \left\| \mathbf{G} \mathbf{w} - \mathbf{b} \right\|_2^2 \quad \text{subject to } \|\mathbf{w}\|_0 \le E_{\text{ECSW}}
\end{equation}
where $G_{e,k} = \text{vech}(\boldsymbol{\Phi}_e^T \mathbf{K}_{e, k} \boldsymbol{\Phi}_e)$ and $b_k = \text{vech}(\mathbf{K}_{r, k})$.

\subsubsection{Key Performance Result}
ECSW guarantees \textbf{symmetric positive-definite (SPD)} reduced stiffness matrices and reduces online matrix assembly time from \textbf{$35.10\text{ ms}$} down to \textbf{$0.90\text{ ms}$} (\textbf{$39\times$ speedup}), enabling 20 FPS real-time Digital Twin execution.
