\section{Governing Dynamic Equations}
To simulate the dynamic response of the parameterized structure, we resolve the semi-discrete second-order system of equations of motion:
\begin{equation}
        \mat{M}(\boldsymbol{\mu}) \ddot{\vect{U}}(t) + \mat{C}(\boldsymbol{\mu}) \dot{\vect{U}}(t) + \mat{K}(\boldsymbol{\mu}) \vect{U}(t) = \lambda(t) \mathbf{F}_{\text{ext}}
\end{equation}
where:
\begin{itemize}
    \item $\mat{M}(\boldsymbol{\mu})$ is the consistent global mass matrix.
    \item $\mat{C}(\boldsymbol{\mu})$ is the viscous damping matrix.
    \item $\mat{K}(\boldsymbol{\mu})$ is the global stiffness matrix.
    \item $\mathbf{F}_{\text{ext}}$ is the spatial profile of the external loads, and $\lambda(t)$ is the time-dependent loading factor.
\end{itemize}

The global consistent mass matrix $\mat{M}$ is assembled from the element mass matrices $\mat{M}_{e}$ computed using Gauss-Legendre quadrature:
\begin{equation}
        \mat{M}_{e, ab} = \int_{-1}^{1} \int_{-1}^{1} \rho R_a(\xi, \eta) R_b(\xi, \eta) |J(\xi, \eta)| \, d\xi \, d\eta
\end{equation}
where $\rho$ is the material mass density. The viscous damping matrix $\mat{C}$ is modeled using the classical Rayleigh formulation:
\begin{equation}
        \mat{C}(\boldsymbol{\mu}) = \alpha_M \mat{M}(\boldsymbol{\mu}) + \beta_K \mat{K}(\boldsymbol{\mu})
\end{equation}
where $\alpha_M$ and $\beta_K$ are the mass-proportional and stiffness-proportional damping factors.

\section{Parameterized Linear Stiffness Matrix Assembly}
In our linear formulation, the parameterized global stiffness matrix $\mat{K}(\boldsymbol{\mu})$ is assembled element-wise using pullback integration over the reference domain $\hat{\Omega}$ from the element stiffness matrices $\mat{K}_e(\boldsymbol{\mu})$:
\begin{equation}
    \mat{K}_{e, ab}(\boldsymbol{\mu}) = \int_{-1}^{1} \int_{-1}^{1} \mat{B}_a^T(\xi, \eta; \boldsymbol{\mu}) \mat{D} \mat{B}_b(\xi, \eta; \boldsymbol{\mu}) |J(\xi, \eta; \boldsymbol{\mu})| \, d\xi \, d\eta
\end{equation}
where $\mat{D}$ is the material constitutive matrix, and the strain-displacement matrix $\mat{B}_a$ depends on the parameters $\boldsymbol{\mu}$ through the coordinate Jacobian transformation:
\begin{equation}
    \mat{B}_a(\xi, \eta; \boldsymbol{\mu}) = \begin{bmatrix} \frac{\partial R_a}{\partial x} & 0 \\[0.3em] 0 & \frac{\partial R_a}{\partial y} \\[0.3em] \frac{\partial R_a}{\partial y} & \frac{\partial R_a}{\partial x} \end{bmatrix}, \quad \text{with} \quad \begin{bmatrix} \frac{\partial R_a}{\partial x} \\[0.3em] \frac{\partial R_a}{\partial y} \end{bmatrix} = \mat{J}^{-1}(\xi, \eta; \boldsymbol{\mu}) \begin{bmatrix} \frac{\partial R_a}{\partial \xi} \\[0.3em] \frac{\partial R_a}{\partial \eta} \end{bmatrix}
\end{equation}
As the notches shape parameter vector $\boldsymbol{\mu} = (r, x_c)^T$ varies, the physical configuration changes, altering the coordinate Jacobian mapping $\mat{J}(\xi, \eta; \boldsymbol{\mu}) = \nabla_{\boldsymbol{\xi}}\boldsymbol{\Psi}(\boldsymbol{\xi}; \boldsymbol{\mu})$ at each Gauss point. This requires updating and re-integrating the stiffness matrix online for every design modification.

\section{Enforcement of Boundary Conditions}
Unlike traditional Lagrangian finite element basis functions, B-spline and NURBS basis functions do not possess the Kronecker delta property at the boundaries (except at the extreme ends of open knot vectors). Thus, control variables $\vect{d}_a$ do not correspond directly to physical displacement values at boundary nodes.

To enforce the homogeneous Dirichlet boundary conditions (e.g., rigid clamping at $x=0$), we employ the \textbf{Penalty Method}. We modify the effective system equations $\mat{K}_{\text{eff}}\vect{U} = \vect{F}_{\text{eff}}$ by adding a large penalty stiffness parameter $k_{\text{p}} \gg 1$ (typically $10^{30}$) to the diagonal entries corresponding to the constrained boundary degrees of freedom:
\begin{equation}
    \mat{K}_{\text{eff}}[i, i] \gets k_{\text{p}}, \quad \vect{F}_{\text{eff}}[i] \gets 0 \quad \text{for all constrained boundary DOFs } i
\end{equation}
This forces the displacement at those boundary degrees of freedom to zero, accurately satisfying the clamping constraint without altering the size or structure of the system matrix.

\section{Implicit Time Integration (\texorpdfstring{Newmark-$\beta$}{Newmark-beta} / Generalized-\texorpdfstring{$\alpha$}{alpha})}
To resolve the linear semi-discrete second-order system of equations of motion, we couple the implicit \textbf{Newmark-$\beta$} average acceleration scheme ($\beta=0.25$, $\gamma=0.5$), which corresponds to the Generalized-$\alpha$ method with high-frequency dissipation spectral radius $\rho_{\infty} = 1.0$ ($\alpha_m = 0$, $\alpha_f = 0$). For a linear system, the equations can be solved directly at each time step without the need for iterative Newton-Raphson solvers. The detailed solver steps are formalized in Algorithm \ref{alg:fom_dynamics}.

\begin{algorithm}[htbp]
\caption{Full Order Model (FOM) Implicit Dynamics Solver}
\label{alg:fom_dynamics}
\begin{algorithmic}[1]
\Procedure{StepDynamicsFOM}{$\vect{U}_n, \dot{\vect{U}}_n, \ddot{\vect{U}}_n, x_c, r, \sigma, \Delta t$}
    \State Assemble mass matrix $\mat{M}(x_c, r)$ and stiffness matrix $\mat{K}(x_c, r)$
    \State Compute Rayleigh damping matrix: $\mat{C} \gets \alpha_M \mat{M} + \beta_K \mat{K}$
    \State Compute external force vector $\vect{F}_{\text{ext}}^{n+1} \gets \text{calculateExternalForceVector}(t_{n+1})$
    \State Set integration constants:
    \State $a_0 = \frac{1}{\beta \Delta t^2}, \quad a_1 = \frac{\gamma}{\beta \Delta t}, \quad a_2 = \frac{1}{\beta \Delta t}, \quad a_3 = \frac{1}{2\beta} - 1, \quad a_4 = \frac{\gamma}{\beta} - 1, \quad a_5 = \Delta t \left( \frac{\gamma}{2\beta} - 1 \right)$
    \State Formulate effective stiffness matrix:
    \State $\mat{K}_{\text{eff}} \gets \mat{K} + a_0 \mat{M} + a_1 \mat{C}$
    \State Formulate effective force vector:
    \State $\vect{F}_{\text{eff}}^{n+1} \gets \vect{F}_{\text{ext}}^{n+1} + \mat{M}(a_0 \vect{U}_n + a_2 \dot{\vect{U}}_n + a_3 \ddot{\vect{U}}_n) + \mat{C}(a_1 \vect{U}_n + a_4 \dot{\vect{U}}_n + a_5 \ddot{\vect{U}}_n)$
    \State Apply penalty Dirichlet clamping at clamped left edge ($x=0$):
    \State $\mat{K}_{\text{eff}}[i, i] \gets 10^{30}, \quad \vect{F}_{\text{eff}}^{n+1}[i] \gets 0$ for all boundary DOFs $i$
    \State Solve system directly: $\vect{U}_{n+1} \gets \text{gaussianElimination}(\mat{K}_{\text{eff}}, \vect{F}_{\text{eff}}^{n+1})$
    \State Update accelerations and velocities:
    \State $\ddot{\vect{U}}_{n+1} \gets a_0 (\vect{U}_{n+1} - \vect{U}_n) - a_2 \dot{\vect{U}}_n - a_3 \ddot{\vect{U}}_n$
    \State $\dot{\vect{U}}_{n+1} \gets \dot{\vect{U}}_n + (1 - \gamma)\Delta t \ddot{\vect{U}}_n + \gamma \Delta t \ddot{\vect{U}}_{n+1}$
    \State \Return $\vect{U}_{n+1}, \dot{\vect{U}}_{n+1}, \ddot{\vect{U}}_{n+1}$
\EndProcedure
\end{algorithmic}
\end{algorithm}

For higher accuracy in transient simulations, we also support the \textbf{generalized-$\alpha$ method}. This scheme extends Newmark-$\beta$ by evaluating the system at intermediate stages $\alpha_m$ and $\alpha_f$ to introduce high-frequency numerical damping while minimizing low-frequency dissipation:
\begin{equation}
    \hat{\mathbf{M}}(\boldsymbol{\mu}) \ddot{\vect{U}}_{t+\alpha_m} + \mat{C}(\boldsymbol{\mu}) \dot{\vect{U}}_{t+\alpha_f} + \mat{K}(\boldsymbol{\mu}) \vect{U}_{t+\alpha_f} = \lambda(t+\alpha_f) \hat{\mathbf{F}}_{ext}
\end{equation}

\section{Quantities of Interest \& Numerical Results}
To evaluate the dynamic performance of the Full Order Model (FOM) under shape variations, we track two primary physical Quantities of Interest (QoI) over the time interval $t \in [0, 10]$ seconds:
\begin{enumerate}
    \item \textbf{Tip Vertical Deflection} $U_y(t)$: Evaluated as the spatial average of the vertical displacement component along the rightmost boundary edge ($x = L$).
    \item \textbf{Maximum Von Mises Stress} $\sigma_{\text{VM}}$: Extracted at the quadrature integration points adjacent to the symmetric top and bottom circular notches to monitor stress concentration factors.
\end{enumerate}

The computational parameters associated with the high-fidelity FOM simulation are summarized as follows:
\begin{itemize}
    \item \textbf{Mesh Discretization}: $E = 3,200$ bivariate elements, corresponding to $N = 6,564$ physical degrees of freedom.
    \item \textbf{Computational Complexity}: Due to the fine mesh discretization and the transient nature of the analysis, solving the full system of equations online at each time step incurs a significant computational load.
\end{itemize}
These numerical complexity considerations highlight the high cost of executing the high-fidelity model directly and justify the implementation of projection-based Reduced Order Modeling in Chapter \ref{ch:rom}.
