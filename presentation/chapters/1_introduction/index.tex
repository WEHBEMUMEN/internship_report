  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Pedagogical Origin to Research Level}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item Started as a \textbf{teaching tool} to illustrate structural dynamics.
          \item Evolved into a \textbf{research framework} for real-time interactive simulations.
        \end{itemize}
      \end{block}
      
      \vspace{0.15cm}
      
      \begin{block}{Objective \& The Problem}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{3pt}
          \item \textbf{The Goal}: Real-time parameter updates for complex shapes.
          \item \textbf{The Bottleneck}: Standard FEM is too slow and requires remeshing.
          \item \textbf{Our Solution}: Exact geometry (IGA) + Fast solvers (ROM).
        \end{itemize}
      \end{block}
    \end{column}
