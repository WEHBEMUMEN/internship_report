      \begin{block}{Pedagogical Origin to Research Level}
        \small
        This project began as a \textbf{pedagogical application} to demonstrate structural dynamics. However, to meet industry demands for interactive simulations, it has been elevated to a rigorous \textbf{research-level} implementation.

        \small
        How to achieve real-time, interactive parameter updates for complex shapes? Standard FEM requires slow online remeshing.
        \begin{itemize}
          \item \textbf{Our Solution}: Exact geometry (IGA) + Fast solvers (ROM).
        \end{itemize}
      \end{block}

% Slide: Conceptual Parallels
\begin{frame}{Conceptual Parallels to Data Compression}
  \begin{block}{IGA \hfill \footnotesize [Hughes et al., 2005]}
    \small
    \begin{itemize}
      \item Exact CAD geometry (NURBS) — smooth boundaries without remeshing.
    \end{itemize}
  \end{block}
  \vspace{0.12cm}
  \begin{block}{ROM \hfill \footnotesize [Benner et al., 2015]}
    \small
    \begin{itemize}
      \item Projects dynamics onto dominant energy-carrying modes (like lossy compression).
      \item Solves equations in reduced space in milliseconds.
    \end{itemize}
  \end{block}
  \vspace{0.12cm}
  \begin{block}{Hyper-Reduction \hfill \footnotesize [Farhat et al., 2014]}
    \small
    \begin{itemize}
      \item Isolates a sparse set of active integration cells (like optimal sensor placement).
      \item Bypasses full-mesh assembly to resolve the online bottleneck.
    \end{itemize}
  \end{block}
\end{frame}
