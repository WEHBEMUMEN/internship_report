% Slide: Conceptual Parallels
\begin{frame}{Conceptual Parallels to Data Compression}
  \begin{block}{IGA vs.\ Tessellation}
    \small
    Standard FEM approximates curved boundaries using piecewise linear tessellations. IGA retains the exact NURBS representation from CAD throughout the analysis — preserving smooth boundary derivatives \textbf{without remeshing}.
  \end{block}
  \vspace{0.12cm}
  \begin{block}{ROM vs.\ Spectral Compression}
    \small
    POD-based ROM projects the system onto the dominant energy-carrying modes — analogous to lossy signal compression. Dynamics are resolved in the reduced space in \textbf{milliseconds}.
  \end{block}
  \vspace{0.12cm}
  \begin{block}{Hyper-Reduction vs.\ Optimal Sensor Arrays}
    \small
    ECSW isolates a sparse set of active integration cells — analogous to deploying an optimized sensor array at critical stress zones to reconstruct the global state while bypassing full-domain computation.
  \end{block}
\end{frame}

% Slide: Research Objectives
\begin{frame}{Research Objectives}
  \begin{enumerate}
    \small
    \item \textbf{Exact Geometry Tracking}: Leverage IGA to represent shape parameter variations of the notched cantilever beam exactly using NURBS control point displacements, maintaining a constant topological mesh structure.
          \vspace{0.05cm}
    \item \textbf{Dimension Reduction}: Extract an optimal, low-dimensional reduced basis using Proper Orthogonal Decomposition (POD) from snapshots of the transient response.
          \vspace{0.05cm}
    \item \textbf{Hyper-Reduction}: Bypass full-mesh assembly for parameter-dependent systems using the Energy-Conserving Sampling and Weighting (ECSW) method to resolve the online computation bottleneck.
          \vspace{0.05cm}
    \item \textbf{Real-Time Implementation}: Demonstrate the feasibility of the proposed ROM in a client-side web application, achieving interactive frame rates for real-time visualization of structural responses.
  \end{enumerate}
\end{frame}
