% Slide: Governing Equations & Strong Form
\begin{frame}{Governing Equations \& Strong Form}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      Assuming small strains and a linear elastic isotropic material, the balance of linear momentum on $\Omega(\boldsymbol{\mu}) \times (0,T]$ is:
      
      \begin{block}{Strong Form}
        \vspace{-0.15cm}
        \[
          \rho\,\ddot{\vect{u}} - \nabla \cdot \boldsymbol{\sigma}(\vect{u}) = \vect{f}_b \quad \text{in } \Omega(\boldsymbol{\mu}) \times (0,T]
        \]
      \end{block}
      
      \par\vspace{0.15cm}
      \textbf{Boundary, Initial \& Constitutive Relations:}
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item $\vect{u} = \vect{0}$ on $\Gamma_D(\boldsymbol{\mu})$ \hfill (Clamped BC)
        \item $\boldsymbol{\sigma}\cdot\vect{n} = \vect{h}_N$ on $\Gamma_N(\boldsymbol{\mu})$ \hfill (Traction BC)
        \item $\vect{u} = \vect{u}_0$, $\dot{\vect{u}} = \vect{v}_0$ at $t=0$ \hfill (Initial Conditions)
        \item $\boldsymbol{\sigma} = \mat{D}:\boldsymbol{\varepsilon}(\vect{u})$ \hfill (Hooke's Law)
        \item $\boldsymbol{\varepsilon}(\vect{u}) = \tfrac{1}{2}\!\left(\nabla\vect{u} + (\nabla\vect{u})^T\right)$ \hfill (Strain tensor)
      \end{itemize}

% Slide: Classical Weak Form
\begin{frame}{Classical Weak Form on $\Omega(\boldsymbol{\mu})$}
  Multiplying the strong form by test functions and integrating by parts yields:
  \begin{block}{Weak Formulation on the Physical Domain}
    \footnotesize
    Find $\vect{u}(t) \in \mathcal{V}(\Omega(\textcolor{red}{\boldsymbol{\mu}}))$ such that:
    \begin{align*}
      &m(\ddot{\vect{u}}, \vect{v}; \textcolor{red}{\boldsymbol{\mu}}) + k(\vect{u}, \vect{v}; \textcolor{red}{\boldsymbol{\mu}}) = f(\vect{v}; \textcolor{red}{\boldsymbol{\mu}}) \quad \forall\,\vect{v}\in\mathcal{V}_0(\Omega(\textcolor{red}{\boldsymbol{\mu}})) \\[4pt]
      \text{where: } &m(\ddot{\vect{u}},\vect{v};\textcolor{red}{\boldsymbol{\mu}}) = \int_{\Omega(\textcolor{red}{\boldsymbol{\mu}})} \rho \ddot{\vect{u}} \cdot \vect{v} \,\mathrm{d}\Omega, \quad k(\vect{u},\vect{v};\textcolor{red}{\boldsymbol{\mu}}) = \int_{\Omega(\textcolor{red}{\boldsymbol{\mu}})} \boldsymbol{\varepsilon}(\vect{v})^T \mat{D} \boldsymbol{\varepsilon}(\vect{u}) \,\mathrm{d}\Omega \\
      &f(\vect{v};\textcolor{red}{\boldsymbol{\mu}}) = \int_{\Omega(\textcolor{red}{\boldsymbol{\mu}})} \vect{f}_b \cdot \vect{v} \,\mathrm{d}\Omega + \int_{\Gamma_N(\textcolor{red}{\boldsymbol{\mu}})} \vect{h}_N \cdot \vect{v} \,\mathrm{d}\Gamma
    \end{align*}
  \end{block}
  \begin{alertblock}{The Parameter Dependency Problem}
    \footnotesize
    \textbf{Space Incompatibility:} Both the domain $\Omega(\textcolor{red}{\boldsymbol{\mu}})$ and spaces $\mathcal{V}(\Omega(\textcolor{red}{\boldsymbol{\mu}}))$ depend on the parameter $\textcolor{red}{\boldsymbol{\mu}}$. \\
    \textbf{Comparison Issue:} Solutions at different parameters cannot be directly compared or linearly combined.
  \end{alertblock}
\end{frame}

% Slide: Pullback Mapping
\begin{frame}{Pullback Mapping to Reference Configuration}
  \small
  To resolve space incompatibility, we map $\Omega(\textcolor{red}{\boldsymbol{\mu}})$ to a fixed reference domain $\hat{\Omega}$ via a geometric diffeomorphism:
  \[
    \vect{x} = \boldsymbol{\Psi}(\hat{\vect{x}};\,\textcolor{red}{\boldsymbol{\mu}}), \quad \mat{J}_{\hat{\vect{x}}}(\hat{\vect{x}};\textcolor{red}{\boldsymbol{\mu}}) = \frac{\partial \boldsymbol{\Psi}}{\partial \hat{\vect{x}}}, \quad J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}} = \det\mat{J}_{\hat{\vect{x}}} > 0
  \]
  
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Main Steps for the Pullback:}
      \footnotesize
      \begin{enumerate}
        \setlength{\itemsep}{3pt}
        \item \textbf{Map functions:} Define $\hat{\vect{u}}(\hat{\vect{x}}) = \vect{u}(\boldsymbol{\Psi}(\hat{\vect{x}};\textcolor{red}{\boldsymbol{\mu}}))$ in parameter-independent space $\mathcal{V}(\hat{\Omega})$.
        \item \textbf{Transform gradients:} $\nabla_{\vect{x}}\vect{u} = \nabla_{\hat{\vect{x}}}\hat{\vect{u}}\,\mat{J}_{\hat{\vect{x}}}^{-1}(\hat{\vect{x}};\textcolor{red}{\boldsymbol{\mu}})$ (chain rule).
        \item \textbf{Change integration variables:} $\mathrm{d}\Omega = J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}}\,\mathrm{d}\hat{\Omega}$ and $\mathrm{d}\Gamma = J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}}^{\Gamma}\,\mathrm{d}\hat{\Gamma}$.
      \end{enumerate}
    \end{column}

% Slide: Pulled-Back Weak Form
\begin{frame}{Pulled-Back Weak Form on $\hat{\Omega}$}
  \small
  Applying the pullback steps to each term in the classical weak form yields:
  
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Classical Form (Physical Domain $\Omega(\boldsymbol{\mu})$)}
        \footnotesize
        Integrals depend implicitly on $\boldsymbol{\mu}$ via the shifting integration domain:
        \[
          m(\ddot{\vect{u}}, \vect{v}; \boldsymbol{\mu}) = \int_{\Omega(\boldsymbol{\mu})} \rho \ddot{\vect{u}} \cdot \vect{v} \,\mathrm{d}\Omega
        \]
        \[
          f(\vect{v}; \boldsymbol{\mu}) = \int_{\Omega(\boldsymbol{\mu})} \vect{f}_b \cdot \vect{v} \,\mathrm{d}\Omega + \int_{\Gamma_N(\boldsymbol{\mu})} \vect{h}_N \cdot \vect{v} \,\mathrm{d}\Gamma
        \]
        \[
          k(\vect{u}, \vect{v}; \boldsymbol{\mu}) = \int_{\Omega(\boldsymbol{\mu})} \boldsymbol{\varepsilon}(\vect{v})^T \mat{D} \boldsymbol{\varepsilon}(\vect{u}) \,\mathrm{d}\Omega
        \]
        \par\vspace{0.1cm}
        \textbf{Limitation:}
        Requires re-meshing and re-integration over a moving domain for every parameter change.
      \end{block}
    \end{column}
    
    \begin{column}{0.49\textwidth}
      \begin{block}{Pulled-Back Form (Reference Domain $\hat{\Omega}$)}
        \footnotesize
        Dependency on $\textcolor{red}{\boldsymbol{\mu}}$ is isolated inside the Jacobian mapping terms:
        \[
          \hat{m}(\ddot{\vect{u}},\delta\vect{u}; \textcolor{red}{\boldsymbol{\mu}}) = \int_{\hat{\Omega}} \rho \ddot{\vect{u}} \cdot \delta\vect{u}\, J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}} \,\mathrm{d}\hat{\Omega}
        \]
        \[
          \hat{f}(\delta\vect{u}; \textcolor{red}{\boldsymbol{\mu}}) = \int_{\hat{\Omega}} \vect{f}_b \cdot \delta\vect{u}\, J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}} \,\mathrm{d}\hat{\Omega} + \int_{\hat{\Gamma}_N} \vect{h}_N \cdot \delta\vect{u}\, J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}}^{\Gamma} \,\mathrm{d}\hat{\Gamma}
        \]
        \[
          \hat{k}(\vect{u},\delta\vect{u}; \textcolor{red}{\boldsymbol{\mu}}) = \int_{\hat{\Omega}} (\mat{J}_{\hat{\vect{x}}}^{-T}\nabla_{\hat{\vect{x}}}\vect{u}):\mat{D}:(\mat{J}_{\hat{\vect{x}}}^{-T}\nabla_{\hat{\vect{x}}}\delta\vect{u})\,J_{\hat{\vect{x}}, \textcolor{red}{\boldsymbol{\mu}}}\,\mathrm{d}\hat{\Omega}
        \]
        \par\vspace{0.1cm}
        \textbf{Advantage:}
        Integration is performed over a static domain; variables are in a parameter-independent space.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
