% Slide: Governing Dynamic Equations (from thesis Ch.4)
\begin{frame}{Governing Dynamic Equations}
  \small
  To simulate the dynamic response we resolve the semi-discrete system:
  \begin{block}{Equation of Motion}
    \vspace{-0.15cm}
    \[
      \mat{M}(\boldsymbol{\mu})\,\ddot{\vect{U}}(t) + \mat{C}(\boldsymbol{\mu})\,\dot{\vect{U}}(t) + \mat{K}(\boldsymbol{\mu})\,\vect{U}(t) = \lambda(t)\,\vect{F}_{\text{ext}}
    \]
    \vspace{-0.45cm}
    \[
      \text{with initial conditions: } \vect{U}(0) = \vect{U}_0, \quad \dot{\vect{U}}(0) = \vect{V}_0
    \]
    \vspace{-0.15cm}
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Mass bilinear form}:
      \[
        m(\vect{v}_h, \ddot{\vect{u}}_h; \boldsymbol{\mu}) = \int_{\Omega(\boldsymbol{\mu})} \rho\,\vect{v}_h \cdot \ddot{\vect{u}}_h\,d\Omega
      \]
      \textbf{Global mass matrix}:
      \[
        \mat{M}_{ab}(\boldsymbol{\mu}) = m(\vect{R}_a, \vect{R}_b; \boldsymbol{\mu})
      \]
      \vspace{-0.2cm}
      \textbf{Rayleigh damping}: $\mat{C}(\boldsymbol{\mu}) = \alpha_M\mat{M}(\boldsymbol{\mu}) + \beta_K\mat{K}(\boldsymbol{\mu})$
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Stiffness bilinear form}:
      \[
        k(\vect{u}_h, \vect{v}_h; \boldsymbol{\mu}) = \int_{\Omega(\boldsymbol{\mu})} \boldsymbol{\varepsilon}(\vect{v}_h)^T \mat{D} \boldsymbol{\varepsilon}(\vect{u}_h) \, d\Omega
      \]
      \textbf{Global stiffness matrix}:
      \[
        \mat{K}_{ab}(\boldsymbol{\mu}) = k(\vect{R}_b, \vect{R}_a; \boldsymbol{\mu})
      \]
      \vspace{-0.3cm}
      \begin{alertblock}{Online Assembly Bottleneck}
        \footnotesize
        As $\boldsymbol{\mu}$ changes, the integration domain $\Omega(\boldsymbol{\mu})$ shifts $\Rightarrow$ full mesh re-assembly is required at each time step.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide: Model Stress & Von Mises Definition
\begin{frame}{Model Stress \& Von Mises Definition}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \small
      \textbf{Stress Reconstruction}:
      \begin{itemize}
        \item \textbf{Key Relation}: Once the displacement field $\vect{u}$ is solved, we evaluate strain $\boldsymbol{\varepsilon}(\vect{u})$, constitutive stress $\boldsymbol{\sigma} = \mat{D}:\boldsymbol{\varepsilon}$, and reconstructed yield criteria.
        \item \textbf{Von Mises Stress $\sigma_{\text{VM}}$}:
              \[
                \sigma_{\text{VM}} = \sqrt{ \sigma_{xx}^2 - \sigma_{xx}\sigma_{yy} + \sigma_{yy}^2 + 3\tau_{xy}^2 }
              \]
        \item \textbf{Stress Concentration}: Localized at notch throats (mechanical fatigue risk).
      \end{itemize}
    \end{column}

% Slide: Mesh Convergence Study
\begin{frame}{Mesh Convergence Study}
  \small
  A dyadic grid convergence study was conducted under static tip bending load using the interactive simulation application to justify the choice of the $80 \times 40$ elements mesh baseline.
  
  \begin{table}[htbp]
    \centering\footnotesize
    \begin{tabular}{lcccc}
      \toprule
      \textbf{Resolution} & \textbf{Elements} & \textbf{DoFs} & \textbf{Tip Disp. $U_y$ (mm)} & \textbf{Relative Error} \\
      \midrule
      $10 \times 5$       & 50                & 168           & 3.538                         & 16.51\% \\
      $20 \times 10$      & 200               & 528           & 3.889                         & 8.24\%  \\
      $40 \times 20$      & 800               & 1,848         & 4.083                         & 3.66\%  \\
      \textbf{80 $\times$ 40} & \textbf{3,200}    & \textbf{6,888} & \textbf{4.185}           & \textbf{1.24\%} \\
      $160 \times 80$     & 12,800            & 26,568        & 4.238                         & Reference \\
      \bottomrule
    \end{tabular}
  \end{table}
  
  The $80 \times 40$ discretization is selected as the operating baseline because:
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Refining to $160 \times 80$ elements yields a change of only $1.24\%$ in displacement.
    \item Avoids a $3.8\times$ increase in equations ($6,888 \to 26,568$ DOFs).
  \end{itemize}
\end{frame}
