% Slide: Governing Dynamic Equations (from thesis Ch.4)
\begin{frame}{Governing Dynamic Equations}
  \small
  To simulate the dynamic response we resolve the semi-discrete system:
  \begin{block}{Equation of Motion}
    \vspace{-0.15cm}
    \[
      \mat{M}(\boldsymbol{\mu})\,\ddot{\vect{U}}(t) + \mat{C}(\boldsymbol{\mu})\,\dot{\vect{U}}(t) + \mat{K}(\boldsymbol{\mu})\,\vect{U}(t) = \lambda(t)\,\vect{F}_{\text{ext}}(\boldsymbol{\mu})
    \]
    \vspace{-0.15cm}
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Global mass matrix}:
      \[
        \mat{M}_{e,ab} = \int_{-1}^1\!\int_{-1}^1 \rho\,R_a R_b\,|J(\xi,\eta)|\,d\xi\,d\eta
      \]
      \vspace{-0.1cm}
      \textbf{Rayleigh damping}: $\mat{C}(\boldsymbol{\mu}) = \alpha_M\mat{M}(\boldsymbol{\mu}) + \beta_K\mat{K}(\boldsymbol{\mu})$
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Stiffness matrix}:
      \[
        \mat{K}_{e,ab}(\boldsymbol{\mu}) = \int_{-1}^{1}\!\int_{-1}^{1}\mat{B}_a^T\mat{D}\mat{B}_b\,|J(\xi,\eta;\boldsymbol{\mu})|\,d\xi\,d\eta
      \]
      \vspace{-0.1cm}
      \begin{alertblock}{Online Assembly Bottleneck}
        \footnotesize
        As $\boldsymbol{\mu}$ changes, $\mat{J}(\xi,\eta;\boldsymbol{\mu})$ varies at every Gauss point $\Rightarrow$ full mesh re-assembly is required at each time step.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide: Newmark-β Integration & Boundary Conditions
\begin{frame}{Implicit Time Integration \& Boundary Conditions}
  \small
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Newmark-$\beta$ Average Acceleration ($\beta=0.25$, $\gamma=0.5$)}
        \footnotesize
        Effective stiffness system per step:
        \[
          \mat{K}_{\text{eff}}\,\vect{U}_{n+1} = \vect{F}_{\text{eff}}^{n+1}
        \]
        where $\mat{K}_{\text{eff}} = \mat{K} + a_0\mat{M} + a_1\mat{C}$ and $a_0 = \tfrac{1}{\beta\Delta t^2}$, $a_1 = \tfrac{\gamma}{\beta\Delta t}$. Unconditionally stable (no Newton loops needed).
      \end{block}
      \vspace{0.05cm}
      \begin{block}{Dirichlet BCs via Penalty Method}
        \footnotesize
        NURBS lack Kronecker delta at boundaries. Enforce clamping by modifying boundary DOFs:
        \[
          \mat{K}_{\text{eff}}[i,i] \gets k_p = 10^{30}, \quad \vect{F}_{\text{eff}}[i] \gets 0
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{alertblock}{FOM Performance}
        \centering\footnotesize
        \begin{tabular}{lr}
          \toprule
          \textbf{Quantity} & \textbf{Value} \\
          \midrule
          Mesh elements & 3,200 \\
          DOFs ($N$)    & 6,564 \\
          Assembly/step & 142.1 ms \\
          Time steps    & 200 \\
          \midrule
          \textbf{Total wall time} & \textbf{28.4 s} \\
          \bottomrule
        \end{tabular}
        \vspace{0.1cm}
        \newline
        \textit{Far too slow for real-time use.}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide: Dynamic Transient Loading Setup
\begin{frame}{Dynamic Transient Loading Setup}
  \small
  To evaluate the structure under dynamic conditions, a transient simulation is executed on the notched cantilever plate over the time interval $t \in [0, 10]$ seconds.
  
  \begin{block}{Excitation Profile}
    \footnotesize
    A time-dependent harmonic bending force is applied at the free tip boundary ($x = L$):
    \[
      \vect{F}(t) = \lambda(t)\,\vect{F}_{\text{ext}} = \sin(\omega t)\,\vect{F}_{\text{ext}} \quad \text{with } \omega = \pi\,\text{rad/s}
    \]
  \end{block}
  \vspace{-0.1cm}
  
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \textbf{Temporal Discretization}:
      \begin{itemize}
        \item Total window: $T = 10$ seconds
        \item Time steps: $200$ ($\Delta t = 0.05$ s)
      \end{itemize}
      \vspace{0.05cm}
      \textbf{Rayleigh Damping Parameters}:
      \[
        \alpha_M = 0.05, \quad \beta_K = 0.001
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \textbf{Quantities of Interest (QoI)}:
      \begin{itemize}
        \item \textbf{Tip Deflection $U_y(t)$}: Average vertical displacement along the rightmost edge.
        \item \textbf{Von Mises Stress $\sigma_{\text{VM}}$}: Monitored at quadrature points adjacent to the notches.
        \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
