% Slide 21: Conclusion & Summary of Thesis Contributions
\begin{frame}{Conclusion \& Summary of Thesis Contributions}
    \textbf{Core Contributions of the Thesis:}
    \begin{enumerate}
        \item Developed a unified framework linking \textbf{Isogeometric Analysis (IGA)} exact boundary CAD geometries with \textbf{Reduced Order Modeling (ROM)}.
        \item Formulated the \textbf{Reference Configuration Pullback} to resolve the shape-parameter mesh update incompatibility.
        \item Applied pre-seeded, dynamically calibrated \textbf{ECSW hyper-reduction} to achieve a \textbf{$64.5\times$ speedup} and \textbf{$0.054\%$} error.
        \item Built and validated a serverless, client-side interactive \textbf{digital twin web application}.
    \end{enumerate}
\end{frame}
