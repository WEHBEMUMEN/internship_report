\begin{frame}{B-Spline Basis Functions}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Over open knot vector $\Xi=\{\xi_0,\ldots,\xi_m\}$, the \textbf{Cox-de Boor recurrence}:
      \[
        N_{i,0}(\xi) = \begin{cases}1 & \xi_i\le\xi<\xi_{i+1}\\0&\text{otherwise}\end{cases}
      \]
      \[
        N_{i,p} = \tfrac{\xi-\xi_i}{\xi_{i+p}-\xi_i}N_{i,p-1} + \tfrac{\xi_{i+p+1}-\xi}{\xi_{i+p+1}-\xi_{i+1}}N_{i+1,p-1}
      \]
      NURBS rational basis with weights $w_i$:
      \[
        R_{i,p}(\xi) = \frac{N_{i,p}(\xi)\,w_i}{\sum_j N_{j,p}(\xi)\,w_j}
      \]
      Physical curve: $\vect{C}(\xi) = \sum_i R_{i,p}(\xi)\,\vect{P}_i$
    \end{column}

% Slide: Isogeometric Paradigm & Mesh
\begin{frame}{The Isogeometric Paradigm \& Mesh}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Key principle}: same NURBS basis for geometry \textit{and} field.
      \begin{block}{Geometry (exact CAD)}
        \vspace{-0.1cm}
        \[
          \vect{x}(\boldsymbol{\xi};\boldsymbol{\mu}) = \sum_{i,j} R_{i,j}(\boldsymbol{\xi})\,\vect{P}_{ij}(\boldsymbol{\mu})
        \]
        \vspace{-0.2cm}
      \end{block}
      \vspace{0.05cm}
      \begin{block}{Field approximation}
        \vspace{-0.1cm}
        \[
          \vect{U}^h(\boldsymbol{\xi},t) = \sum_{a} R_a(\boldsymbol{\xi})\,\vect{d}_a(t)
        \]
        \vspace{-0.2cm}
      \end{block}
      \vspace{0.05cm}
      \begin{exampleblock}{Chosen Discretization}
        $80\times40$ elements, degree $p=2$,\\
        $N=6{,}564$ DOFs, $E=3{,}200$ elements
      \end{exampleblock}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{alertblock}{Why IGA for Shape ROM?}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{2pt}
          \item $C^{p-1}$ continuity — superior accuracy/DOF vs.\ $C^0$ FEM
          \item Shape change = control point update only (mesh topology remains fixed)
          \item Parametric space $[0,1]^2 \equiv$ reference domain $\hat{\Omega}$
          \item No remeshing — exact CAD boundary at all parameter values
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide: h-Refinement (Knot Insertion)
\begin{frame}{$h$-Refinement (Knot Insertion)}
  \small
  New elements (knot spans) are created by inserting new values into the knot vectors without modifying the polynomial degree $p$ or the geometric shape.
  
  \begin{block}{Boehm's Knot Insertion Algorithm}
    \footnotesize
    For a new knot $\bar{\xi}$, the refined control points $\vect{P}_i^{\text{new}}$ are computed locally:
    \[
      \vect{P}_i^{\text{new}} = \alpha_i \vect{P}_i + (1 - \alpha_i) \vect{P}_{i-1}
    \]
    where the blending coefficients $\alpha_i$ are determined by the knot layout:
    \[
      \alpha_i = \text{1 (if } i \le k - p), \quad \tfrac{\bar{\xi} - \xi_i}{\xi_{i+p} - \xi_i} \text{ (if } k - p + 1 \le i \le k), \quad \text{0 (if } i \ge k + 1)
    \]
  \end{block}
  \vspace{-0.1cm}
  For a 2D control lattice grid $\mat{P}$, refinement is applied tensorially:
  \[
    \vect{P}^{\text{new}} = \left( \mat{T}_{\eta} \otimes \mat{T}_{\xi} \right) \vect{P}, \quad \text{where } \mat{P}^{\text{new}} = \mat{T}_{\xi} \mat{P} \mat{T}_{\eta}^T
  \]
  Refined bases retain $C^{p-1}$ continuity at the inserted knot coordinates.
\end{frame}

% Slide: p-Refinement (Degree Elevation)
\begin{frame}{$p$-Refinement (Degree Elevation)}
  \small
  Polynomial degree is elevated from $p$ to $p + \Delta p$ without changing the geometry or the parametric coordinates of existing knot spans.
  
  \begin{block}{Linear Operator Mapping}
    \footnotesize
    The new elevated control points are related to the original grid by a linear operator:
    \[
      \vect{P}^{\text{new}} = \mat{T}_p \vect{P}
    \]
    To preserve the geometric boundary shape and continuity properties, the multiplicity of all existing knots must be increased by $\Delta p$.
  \end{block}
  \vspace{-0.1cm}
  \begin{exampleblock}{Advantages of IGA Degree Elevation}
    \footnotesize
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item Adds degrees of freedom and expands basis support.
      \item Preserves the non-negativity and variation-diminishing properties.
      \item Avoids high-order oscillations (Runge's phenomenon) common in classical Lagrange-based FEM $p$-refinement.
    \end{itemize}
  \end{exampleblock}
\end{frame}

% Slide: k-Refinement (Order Elevation)
\begin{frame}{$k$-Refinement (Order Elevation)}
  \small
  Unique to IGA (no analogue in classical FEA). Allows increasing polynomial degree while maintaining maximum inter-element continuity.
  
  \begin{block}{Sequence of Operations (Non-Commutative)}
    \footnotesize
    \begin{enumerate}
      \setlength{\itemsep}{1pt}
      \item \textbf{Degree Elevation}: Elevate degree from $p$ to $p + \Delta p$ on the coarse boundary-only knot vector.
      \item \textbf{Knot Insertion}: Insert internal knots into the elevated vector.
    \end{enumerate}
  \end{block}
  \vspace{-0.1cm}
  \begin{alertblock}{Continuity Advantage}
    \footnotesize
    Degree elevation first followed by knot insertion yields basis functions with $\mathbf{C^{p + \Delta p - 1}}$ continuity at all internal element boundaries. Provides superior accuracy per degree of freedom for dynamics.
  \end{alertblock}
\end{frame}

    \item Avoids a $3.8\times$ increase in equations ($6,888 \to 26,568$ DOFs).
  \end{itemize}
