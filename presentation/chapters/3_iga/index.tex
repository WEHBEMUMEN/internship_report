% Slide: Why IGA?
\begin{frame}{Why IGA?}
  \small
  IGA bridges the gap between CAD and FEA by using CAD basis functions for analysis:
  
  \begin{itemize}
    \setlength{\itemsep}{0.4cm}
    \item \textbf{Direct CAD Integration:} Runs simulations directly on NURBS geometry, bypassing time-consuming mesh generation.
    \item \textbf{Exact Geometry:} Preserves exact CAD boundaries throughout the analysis, eliminating geometric errors.
    \item \textbf{Higher Continuity ($C^{p-1}$):} Provides high regularity across element boundaries for superior accuracy per DOF compared to classical $C^0$ FEM.
  \end{itemize}
\end{frame}

% Slide: Core Ingredients of IGA
\begin{frame}{The Core Ingredients of IGA}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Knot Vector}
        \begin{minipage}[t][3.0cm]{\textwidth}
        \scriptsize
        Ordered set of parametric coordinates:
        \[
          \Xi = \{\xi_0, \xi_1, \dots, \xi_m\}
        \]
        Non-zero knot spans partition the parametric domain into elements.
        \end{minipage}
      \end{block}
      \vspace{0.1cm}
      \begin{block}{3. Weights (NURBS)}
        \begin{minipage}[t][3.0cm]{\textwidth}
        \scriptsize
        Weights $w_i$ rationalise the basis functions to construct NURBS:
        \[
          R_{i,p}(\xi) = \frac{N_{i,p}(\xi)w_i}{\sum_j N_{j,p}(\xi)w_j}
        \]
        Allows exact representation of conic shapes (circles, cylinders).
        \end{minipage}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{2. B-Spline Basis}
        \begin{minipage}[t][3.0cm]{\textwidth}
        \scriptsize
        Polynomial basis functions constructed recursively over $\Xi$ via Cox-de Boor:
        \[
          N_{i,0}(\xi) = \begin{cases} 1 & \xi_i \le \xi < \xi_{i+1} \\ 0 & \text{otherwise} \end{cases}
        \]
        Yields $C^{p-1}$ continuity across internal element boundaries.
        \end{minipage}
      \end{block}
      \vspace{0.1cm}
      \begin{block}{4. Control Points}
        \begin{minipage}[t][3.0cm]{\textwidth}
        \scriptsize
        Physical coordinates $\vect{P}_i$ that parameterize the geometry:
        \[
          \vect{C}(\xi) = \sum_i R_{i,p}(\xi)\,\vect{P}_i
        \]
        Read directly from CAD formats (\textbf{`.stp`}, \textbf{`.igs`}), avoiding standard FE mesh generation.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide: B-Spline Basis Functions — figure copied from thesis Ch.3
\begin{frame}{B-Spline Basis Functions}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Over open knot vector $\Xi=\{\xi_0,\ldots,\xi_m\}$, the \textbf{Cox-de Boor recurrence}:
      \[
        N_{i,0}(\xi) = \begin{cases}1 & \xi_i\le\xi<\xi_{i+1}\\0&\text{otherwise}\end{cases}
      \]
      \[
        N_{i,p} = \tfrac{\xi-\xi_i}{\xi_{i+p}-\xi_i}N_{i,p-1} + \tfrac{\xi_{i+p+1}-\xi}{\xi_{i+p+1}-\xi_{i+1}}N_{i+1,p-1}
      \]
      \par\vspace{0.2cm}
      \textbf{Key Properties:}
      \begin{itemize}
        \scriptsize
        \item \textbf{Partition of Unity:} $\sum_{i} N_{i,p}(\xi) = 1 \quad \forall\,\xi$
        \item \textbf{Non-negativity:} $N_{i,p}(\xi) \ge 0 \quad \forall\,\xi, i$
        \item \textbf{Local Support:} $N_{i,p}(\xi) = 0$ outside $[\xi_i, \xi_{i+p+1})$
      \end{itemize}
    \end{column}

\begin{frame}{B-Spline Curve Concept}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \small
      A B-spline/NURBS curve $\vect{C}(\xi)$ is defined as:
      \[
        \vect{C}(\xi) = \sum_{i=0}^{n-1} R_{i,p}(\xi) \vect{P}_i
      \]
      \begin{itemize}
        \item $\vect{P}_i$ represent the \textbf{control points}.
        \item The control points form a piecewise linear \textbf{control polygon}.
        \item Lack of Kronecker delta property: $\vect{C}(\xi_i) \neq \vect{P}_i$ in general, except at the clamped boundary ends.
      \end{itemize}

\begin{frame}{2D Bivariate IGA Mapping}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \textbf{Tensor-Product NURBS}:
      \begin{itemize}
        \item Bivariate rational basis functions $R_{i,j}^{p,q}(\xi, \eta)$ extend 1D B-splines to 2D surfaces:
        \[
          R_{i,j}^{p,q}(\xi, \eta) = \frac{N_{i,p}(\xi) M_{j,q}(\eta) w_{i,j}}{\sum_{k} \sum_{l} N_{k,p}(\xi) M_{l,q}(\eta) w_{k,l}}
        \]
        \item Geometric mapping $\vect{S}(\xi, \eta)$ maps the parametric square $[0,1]^2$ to the physical 2D domain:
        \[
          \vect{S}(\xi, \eta) = \sum_{i} \sum_{j} R_{i,j}^{p,q}(\xi, \eta) \vect{P}_{i,j}
        \]
        \item $\vect{P}_{i,j}$ forms a bidimensional control net lattice.
      \end{itemize}

    \begin{column}{0.48\textwidth}
      \small
      \textbf{Knot Insertion}: Adding elements by subdividing knot spans.
      \par\vspace{0.1cm}

        \item \textbf{Why}: Localize refinement near high-stress areas (e.g., notches).

% Slide: p-Refinement (Degree Elevation)
\begin{frame}{$p$-Refinement (Degree Elevation)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \textbf{Degree Elevation}: Increasing polynomial degree of basis functions.
      \par\vspace{0.15cm}
      \begin{itemize}
        \item \textbf{Why}: Boost accuracy for smooth solutions and wave dynamics.
        \item \textbf{How}: Linear operator elevates degree $p \to p + \Delta p$, increasing knot multiplicity to preserve continuity.
        \item \textbf{Pros}: Keeps geometric shape identical. Avoids high-order Runge oscillations of Lagrange FEM.
        \item \textbf{Cons}: Wider basis support increases matrix bandwidth and solver overhead.
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering
      \scalebox{0.72}{
      \begin{tikzpicture}[scale=0.85, >=stealth]

\begin{frame}{$k$-Refinement \& Exact Geometry}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \textbf{k-Refinement}: Elevating polynomial degree while maintaining maximum inter-element continuity.
      \par\vspace{0.15cm}
      \begin{itemize}
        \item \textbf{Why}: Achieve maximum accuracy per DOF. High continuity ($C^{p-1}$) is optimal for dynamics.
        \item \textbf{How}: Elevate degree first ($p \to p+\Delta p$), \textit{then} insert internal knots.
        \item \textbf{Pros}: Unique to IGA. Keeps DOFs minimal for a given error tolerance.
        \item \textbf{Cons}: Higher computational assembly stencil (wider support).
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{alertblock}{Continuity Comparison}
        \centering\scriptsize
        \begin{tabular}{lcc}
          \toprule
          \textbf{Method} & \textbf{Sequence} & \textbf{Continuity} \\
          \midrule
          $p \to h$ & Insert first & $C^0$ (FEA-like) \\
          \textbf{$k$-ref} & \textbf{Elevate first} & \textbf{$C^{p+\Delta p - 1}$} \\
          \bottomrule
        \end{tabular}

\begin{frame}{Mesh Convergence Study}
  \small
  A dyadic grid convergence study was conducted under static tip bending load using the interactive simulation application to justify the choice of the $80 \times 40$ elements mesh baseline.
  
  \begin{table}[htbp]
    \centering\footnotesize
    \begin{tabular}{lcccc}
      \toprule
      \textbf{Resolution} & \textbf{Elements} & \textbf{DoFs} & \textbf{Tip Disp. $U_y$ (mm)} & \textbf{Relative Error} \\
      \midrule

      $20 \times 10$      & 200               & 528           & 3.889                         & 8.24\%  \\
      $40 \times 20$      & 800               & 1,848         & 4.083                         & 3.66\%  \\
      \textbf{80 $\times$ 40} & \textbf{3,200}    & \textbf{6,888} & \textbf{4.185}           & \textbf{1.24\%} \\
      $160 \times 80$     & 12,800            & 26,568        & 4.238                         & Reference \\
      \bottomrule
    \end{tabular}
  \end{table}
  
  The $80 \times 40$ discretization is selected as the operating baseline because:
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Refining to $160 \times 80$ elements yields a change of only $1.24\%$ in displacement.
    \item Avoids a $3.8\times$ increase in equations ($6,888 \to 26,568$ DOFs).
  \end{itemize}
\end{frame}
