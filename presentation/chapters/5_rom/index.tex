% Frame 1: Why ROM? (Motivation)
% ==========================================
\begin{frame}{Why ROM? (Motivation)}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{The Full-Order Bottleneck}
        \begin{itemize}
          \item High-fidelity Full Order Model (FOM) has $N = 6,564$ DOFs.
          \item Online matrix assembly scales with mesh size $\mathcal{O}(N)$.
          \item Solves are too slow for real-time interactivity.
        \end{itemize}
      \end{block}
      \vspace{0.1cm}
      \begin{block}{Need for Interactivity}
        \begin{itemize}
          \item Interactive client-side web browser (Digital Twin).
          \item Must bypass the full-order assembly bottleneck.
          \item Target frame rate: 20--60 FPS ($<1.5\text{ ms}$ solve time).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{alertblock}{Interactive Budget}
        \centering
        \bfseries
        Online Solve Time: \Large $< 1.5\text{ ms}$
      \end{alertblock}

% Frame 3: Step 1: Snapshot Collection

\begin{frame}{Step 1: Snapshot Collection (Data Generation)}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{What is a Snapshot?}
        \begin{itemize}
          \item A high-fidelity state vector representing the structural displacement field $\vect{U}(t) \in \mathbb{R}^N$ at a given time or parameter set.
        \end{itemize}
      \end{block}
      \begin{block}{Compilation of snap matrix}
        \begin{itemize}
          \item Compile all state vectors into snapshot matrix:
          \[
            \mathbf{S}_u = [\vect{u}_1, \vect{u}_2, \dots, \vect{u}_d] \in \mathbb{R}^{N \times d}
          \]
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{Representative Load Cases}
        FOM simulations are run offline across representative configurations to capture dominant physics:
        \begin{itemize}
          \item Tip bending load
          \item Mid-span bending load
          \item Analytical axial tension
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% ==========================================
% Frame 4: Step 2: SVD / POD Basis Construction
% ==========================================
\begin{frame}{Step 2: SVD / POD Basis Construction}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Dimensionality Reduction}:
      \begin{itemize}
        \item Perform Singular Value Decomposition (SVD):
        \[
          \mathbf{S}_u = \mathbf{U} \boldsymbol{\Sigma} \mathbf{V}^T
        \]
        \item Keep the first $r$ dominant modes of $\mathbf{U}$ to form the Reduced Basis matrix $\boldsymbol{\Phi} \in \mathbb{R}^{N \times r}$:
        \[
          \vect{U}(t) \approx \boldsymbol{\Phi}\,\vect{q}(t), \quad r\ll N
        \]
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{POD Convergence Metrics}
        \footnotesize
        \centering
        \begin{tabular}{lcc}
          \toprule
          Modes $r$ & Error $\epsilon(r)$ & Energy $\eta(r)$ \\
          \midrule
          1  & 2.34\% & 99.945\% \\
          5  & 0.89\% & 99.992\% \\
          10 & 0.11\% & 99.999\% \\
          \bottomrule
        \end{tabular}
        \vspace{0.1cm}
        \scriptsize
        $\epsilon(r) = \sqrt{1-\eta(r)}$ decays rapidly, reaching high accuracy at just $r=10$ modes.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ==========================================
% Frame 5: Step 3: Reference Pullback Mapping
% ==========================================
\begin{frame}{Step 3: Reference Pullback Mapping}
  \small
  \begin{block}{Diffeomorphic Pullback to Static $\hat{\Omega}$ (Thesis Section 3.4)}
    \footnotesize
    Coordinate transformation maps varying physical coordinates $\vect{x} \in \Omega(\boldsymbol{\mu})$ to static reference coordinates $\hat{\vect{x}} \in \hat{\Omega}$:
    \[
      \nabla_{\vect{x}} \vect{u} = \mat{J}^{-T}(\boldsymbol{\xi}; \boldsymbol{\mu}) \nabla_{\boldsymbol{\xi}} \hat{\vect{u}}, \quad d\Omega = \color{cnamred}{J_{\boldsymbol{\mu}}(\boldsymbol{\xi})} \, d\hat{\Omega}
    \]
    where $\mat{J}(\boldsymbol{\xi}; \boldsymbol{\mu}) = \frac{\partial \vect{x}}{\partial \hat{\vect{x}}}$ is the coordinate Jacobian, and $\color{cnamred}{J_{\boldsymbol{\mu}}(\boldsymbol{\xi}) = \det\mat{J}}$ is the mapping weight.
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{State Projection on Static Basis ($\boldsymbol{\Phi}$)}
        \scriptsize
        The discrete state vector $\vect{U}(\boldsymbol{\mu})$ is parameter-dependent, but projected onto static POD modes:
        \[
          \vect{U}(\boldsymbol{\mu}) \approx \boldsymbol{\Phi} \vect{q}(\boldsymbol{\mu})
        \]
        \textbf{Why this matters}:
        \begin{itemize}
          \item \textbf{Static Modes}: The SVD projection matrix $\boldsymbol{\Phi}$ is constant and computed offline.
          \item \textbf{Online Efficiency}: Only the low-dimensional state vector $\vect{q}(\boldsymbol{\mu}) \in \mathbb{R}^r$ needs to be resolved online.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{The Parameter Dependency ($\mat{K}(\boldsymbol{\mu})$)}
        \scriptsize
        Stiffness bilinear form maps parameter $\boldsymbol{\mu}$ into the spatial integral:
        \[
          \hat{k}(\vect{u}, \delta\vect{u}; \boldsymbol{\mu}) = \int_{\hat{\Omega}} \left(\mat{J}^{-T} \nabla_{\boldsymbol{\xi}}\vect{u}\right) : \mat{D} : \left(\mat{J}^{-T} \nabla_{\boldsymbol{\xi}}\delta\vect{u}\right) \color{cnamred}{J_{\boldsymbol{\mu}}} \, d\hat{\Omega}
        \]
        \textbf{Why it matters}:
        \begin{itemize}
          \item Traps $\boldsymbol{\mu}$ inside the integral.
          \item Standard ROM cannot precompute $\mat{\Phi}^T \mat{K}(\boldsymbol{\mu})\mat{\Phi}$.
          \item Requires $O(N)$ full online assembly.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% Frame 6: Step 4: Galerkin Projection vs. ECSW Hyper-reduction
% ==========================================
\begin{frame}{Step 4: Galerkin Projection vs. ECSW Hyper-reduction}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Galerkin Projection ($r \times r$)}
        \footnotesize
        \[
          \mat{M}_r\ddot{\vect{q}} + \mat{C}_r\dot{\vect{q}} + \mat{K}_r(\boldsymbol{\mu})\vect{q} = \vect{F}_r(t)
        \]
        \alert{Assembly Bottleneck}: Standard Galerkin still requires assembling the full matrix $\hat{\mat{K}}(\boldsymbol{\mu}) \in \mathbb{R}^{N \times N}$ online. No online speedup ($0.92\times$).
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{ECSW Hyper-Reduction}
        \footnotesize
        ECSW integrates only over a sparse active subset of elements $\hat{\Omega} \subset \Omega$:
        \[
          \mat{K}_{r,\text{ECSW}}(\boldsymbol{\mu}) = \sum_{e\in\hat{\Omega}} w_e\,\boldsymbol{\Phi}_e^T\mat{K}_e(\boldsymbol{\mu})\boldsymbol{\Phi}_e
        \]
        Bypasses the assembly bottleneck, evaluating only $28$ out of $3,200$ elements online.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Frame 8: Client-Side Software Architecture
% ==========================================
\begin{frame}{Client-Side Software Architecture (Digital Twin)}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      The Digital Twin is a \textbf{purely client-side web application} (no server-side backend).
      \vspace{0.05cm}
      \begin{block}{Model-View-Controller (MVC)}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{1pt}
          \item \textbf{Model}: IGA-ROM engine executing solver loops using \texttt{Float64Array} typed arrays.
          \item \textbf{View}: HTML5 Canvas rendering Von Mises stress heatmap.
          \item \textbf{Controller}: Direct binding from UI sliders to solver parameter variables.
        \end{itemize}
      \end{block}
      \vspace{0.05cm}
      ROM datasets ($\boldsymbol{\Phi}$, $w_e$) loaded dynamically.
    \end{column}
